# DS18B20_MQTT - XIAO ESP32-S3 : plan d'implémentation

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:executing-plans to implement this plan task-by-task.

**Goal:** Publier la température d'une sonde DS18B20 branchée sur une XIAO ESP32S3 dans Home Assistant via MQTT Discovery.

**Architecture:** Projet PlatformIO. La logique pure (validation des lectures, arrondi, décision de publier, état de la sonde, sujets et JSON de découverte) vit dans `lib/temp_logic/`, sans Arduino, testée en natif avec Unity. `src/main.cpp` ne fait que le matériel et le réseau : 1-Wire, WiFi, MQTT.

**Tech Stack:** pioarduino (Arduino sur ESP-IDF 5), OneWire 2.3.8, DallasTemperature 4.x, PubSubClient 2.8, ArduinoJson 7, Unity, cppcheck.

```cpp
#include "temp_logic.h"

#include <math.h>
#include <stdio.h>
#include <stdlib.h>

namespace temp {

namespace {
constexpr float kDisconnected = -127.0f;
constexpr float kPowerOn = 85.0f;
constexpr float kPowerOnTolerance = 10.0f;
constexpr float kMin = -55.0f;
constexpr float kMax = 125.0f;
}  // namespace

bool isValidReading(float celsius, bool hasPrevious, float previous) {
    if (celsius == kDisconnected) return false;
    if (celsius < kMin || celsius > kMax) return false;
    if (celsius == kPowerOn) {
        return hasPrevious && fabsf(previous - kPowerOn) <= kPowerOnTolerance;
    }
    return true;
}

int toTenths(float celsius) {
    return (int)lroundf(celsius * 10.0f);
}

void formatTenths(int tenths, char *out, size_t size) {
    const int a = abs(tenths);
    snprintf(out, size, "%s%d.%d", tenths < 0 ? "-" : "", a / 10, a % 10);
}

}  // namespace temp
```

Dans `temp_logic.h` :

```cpp
// Publie la premiere valeur, puis a chaque ecart >= minDeltaTenths avec la
// derniere publiee, ou quand maxIntervalMs s'est ecoule depuis (heartbeat).
class PublishPolicy {
  public:
    explicit PublishPolicy(int minDeltaTenths = 2, uint32_t maxIntervalMs = 300000)
        : minDelta_(minDeltaTenths), maxInterval_(maxIntervalMs) {}

    bool shouldPublish(int tenths, uint32_t nowMs) const;
    void markPublished(int tenths, uint32_t nowMs);
    // Oublie la derniere publication (reconnexion MQTT) : la suivante part.
    void reset() { hasPublished_ = false; }

  private:
    int minDelta_;
    uint32_t maxInterval_;
    bool hasPublished_ = false;
    int lastTenths_ = 0;
    uint32_t lastMs_ = 0;
};
```

Dans `temp_logic.cpp` :

```cpp
bool PublishPolicy::shouldPublish(int tenths, uint32_t nowMs) const {
    if (!hasPublished_) return true;
    if (abs(tenths - lastTenths_) >= minDelta_) return true;
    return (uint32_t)(nowMs - lastMs_) >= maxInterval_;
}

void PublishPolicy::markPublished(int tenths, uint32_t nowMs) {
    hasPublished_ = true;
    lastTenths_ = tenths;
    lastMs_ = nowMs;
}
```

```cpp
class ProbeHealth {
  public:
    explicit ProbeHealth(uint8_t maxFailures = 3) : maxFailures_(maxFailures) {}

    // Enregistre une lecture ; renvoie true si l'etat en ligne a change.
    bool record(bool valid);
    bool online() const { return online_; }

  private:
    uint8_t maxFailures_;
    uint8_t failures_ = 0;
    bool online_ = false;
};
```

Dans `temp_logic.cpp` :

```cpp
bool ProbeHealth::record(bool valid) {
    const bool before = online_;
    if (valid) {
        failures_ = 0;
        online_ = true;
    } else if (failures_ < maxFailures_ && ++failures_ >= maxFailures_) {
        online_ = false;
    }
    return online_ != before;
}
```

```cpp
#pragma once

#include <stddef.h>
#include <stdint.h>

namespace temp {

struct Topics {
    char state[64];
    char status[64];
    char probe[64];
    char config[96];
};

// "xiao_ds18b20_" + 3 derniers octets de la MAC en hexa minuscule.
void makeDeviceId(const uint8_t mac[6], char *out, size_t size);

void makeTopics(const char *deviceId, Topics &t);

// JSON de decouverte du capteur de temperature. Renvoie la longueur ecrite,
// 0 si le tampon est trop petit.
size_t buildDiscovery(const char *deviceId, const Topics &t, char *out, size_t size);

}  // namespace temp
```

```cpp
#include "ha_discovery.h"

#include <ArduinoJson.h>
#include <stdio.h>

namespace temp {

void makeDeviceId(const uint8_t mac[6], char *out, size_t size) {
    snprintf(out, size, "xiao_ds18b20_%02x%02x%02x", mac[3], mac[4], mac[5]);
}

void makeTopics(const char *deviceId, Topics &t) {
    snprintf(t.state, sizeof t.state, "xiao_ds18b20/%s/temperature", deviceId);
    snprintf(t.status, sizeof t.status, "xiao_ds18b20/%s/status", deviceId);
    snprintf(t.probe, sizeof t.probe, "xiao_ds18b20/%s/probe", deviceId);
    snprintf(t.config, sizeof t.config,
             "homeassistant/sensor/%s/temperature/config", deviceId);
}

size_t buildDiscovery(const char *deviceId, const Topics &t, char *out, size_t size) {
    char uniqueId[48];
    snprintf(uniqueId, sizeof uniqueId, "%s_temperature", deviceId);

    JsonDocument doc;
    doc["name"] = "Temperature";
    doc["unique_id"] = uniqueId;
    doc["state_topic"] = t.state;
    doc["device_class"] = "temperature";
    doc["state_class"] = "measurement";
    doc["unit_of_measurement"] = "°C";
    doc["suggested_display_precision"] = 1;
    doc["availability_mode"] = "all";
    JsonArray avail = doc["availability"].to<JsonArray>();
    avail.add<JsonObject>()["topic"] = t.status;
    avail.add<JsonObject>()["topic"] = t.probe;
    JsonObject dev = doc["device"].to<JsonObject>();
    dev["identifiers"].add(deviceId);
    dev["name"] = deviceId;
    dev["model"] = "XIAO ESP32S3 + DS18B20";
    dev["manufacturer"] = "Seeed Studio";

    if (measureJson(doc) >= size) return 0;
    return serializeJson(doc, out, size);
}

}  // namespace temp
```

```cpp
    }
    const bool ok = mqtt.publish(topics.config, (const uint8_t *)payload, n, true);
    Serial.printf("Decouverte publiee (%u octets) : %s\n", (unsigned)n, ok ? "OK" : "ECHEC");
}

static void onMqttMessage(char *topic, byte *payload, unsigned int length) {
    if (strcmp(topic, HA_STATUS_TOPIC) == 0 && length == 6 &&
        memcmp(payload, "online", 6) == 0) {
        discoveryRequested = true;  // Home Assistant a redemarre
    }
}

static void handleWifi() {
    static uint32_t lastAttempt = 0;
    static bool wasConnected = false;
    const bool connected = WiFi.status() == WL_CONNECTED;
    if (connected != wasConnected) {
        wasConnected = connected;
        if (connected) {
            Serial.printf("WiFi connecte a %s, IP %s\n", WiFi.SSID().c_str(),
                          WiFi.localIP().toString().c_str());
        } else {
            Serial.println("WiFi perdu");
        }
    }
    if (connected) return;
    const uint32_t now = millis();
    if (now - lastAttempt < RETRY_INTERVAL_MS) return;
    lastAttempt = now;
    wifiMulti.run(3000);  // ne bloque que tant que le WiFi est absent
}

static void handleMqtt() {
    static uint32_t lastAttempt = 0;
    if (WiFi.status() != WL_CONNECTED) return;
    if (mqtt.connected()) {
        mqtt.loop();
        if (discoveryRequested) {
            discoveryRequested = false;
            publishDiscovery();
        }
        return;
    }
    const uint32_t now = millis();
    if (now - lastAttempt < RETRY_INTERVAL_MS) return;
    lastAttempt = now;
    if (!mqtt.connect(deviceId, MQTT_USER, MQTT_PASS, topics.status, 0, true, "offline")) {
        Serial.printf("MQTT : echec de connexion (etat %d)\n", mqtt.state());
        return;
    }
    Serial.println("MQTT connecte");
    mqtt.publish(topics.status, "online", true);
    publishProbeStatus();
    publishDiscovery();
    mqtt.subscribe(HA_STATUS_TOPIC);
    policy.reset();  // republier la valeur courante a la prochaine lecture
}

static void publishTemperature(int tenths) {
    char value[8];
    temp::formatTenths(tenths, value, sizeof value);
    if (mqtt.connected() && policy.shouldPublish(tenths, millis())) {
        if (mqtt.publish(topics.state, value)) {
            policy.markPublished(tenths, millis());
            Serial.printf("Publie : %s C\n", value);
        }
    }
}

static void handleSensor() {
    const uint32_t now = millis();
    if (!converting) {
        if (now - lastRead < READ_INTERVAL_MS && lastRead != 0) return;
        sensors.requestTemperatures();
        converting = true;
        conversionStart = now;
        return;
    }
    if (now - conversionStart < sensors.millisToWaitForConversion()) return;
    converting = false;
    lastRead = now;

    const float c = sensors.getTempCByIndex(0);
    const bool valid = temp::isValidReading(c, hasPrevious, previous);
    Serial.printf("Lecture : %.4f C (%s)\n", c, valid ? "valide" : "rejetee");
    if (probe.record(valid)) {
        Serial.printf("Sonde %s\n", probe.online() ? "en ligne" : "hors ligne");
        if (mqtt.connected()) publishProbeStatus();
    }
    if (!valid) return;
    hasPrevious = true;
    previous = c;
    publishTemperature(temp::toTenths(c));
}

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("DS18B20_MQTT - XIAO ESP32-S3");

    // Pas de resistance de tirage externe : pull-up interne (~45 kohm).
    pinMode(ONE_WIRE_PIN, INPUT_PULLUP);
    sensors.begin();
    sensors.setResolution(12);
    sensors.setWaitForConversion(false);
    Serial.printf("Sondes detectees : %u\n", sensors.getDeviceCount());

    WiFi.mode(WIFI_STA);
    uint8_t mac[6];
    WiFi.macAddress(mac);
    temp::makeDeviceId(mac, deviceId, sizeof deviceId);
    temp::makeTopics(deviceId, topics);
    Serial.printf("Identifiant : %s\n", deviceId);

    wifiMulti.addAP(WIFI_SSID, WIFI_PASSWORD);
#ifdef WIFI_SSID_2
    wifiMulti.addAP(WIFI_SSID_2, WIFI_PASSWORD_2);
#endif

    mqtt.setServer(MQTT_SERVER, MQTT_PORT);
    mqtt.setBufferSize(1024);  // decouverte > 256 octets par defaut
    mqtt.setCallback(onMqttMessage);
}

void loop() {
    handleWifi();
    handleMqtt();
    handleSensor();
    delay(10);
}
```

Points d'attention :
- `pinMode(INPUT_PULLUP)` **avant** `sensors.begin()` : OneWire ne touche ensuite qu'à l'activation de la sortie, le pull-up reste.
- `getTempCByIndex(0)` refait une recherche sur le bus : une sonde branchée après le démarrage est prise en compte, avec sa résolution d'usine de 12 bits.
- `D0` est défini par la variante XIAO_ESP32S3 du core (GPIO1). Si la compilation échoue sur `D0`, remplacer par `1`.
